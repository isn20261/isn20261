    **7) Sanitization helper.** Strip stack traces, Authorization headers (defensive — backend should never echo them, but treat the response as untrusted), and clip overly long messages. Apply a default if the result is empty.

    ```ts
    const MAX_USER_MESSAGE_LEN = 240;

    function sanitizeMessage(raw: unknown, fallback: string): string {
      if (typeof raw !== "string" || raw.trim() === "") return fallback;
      // Remove anything that looks like a header, a stack frame, or a path on the FS.
      const stripped = raw
        .replace(/Authorization:\s*Bearer\s+[A-Za-z0-9._-]+/gi, "[redacted]")
        .replace(/\bat\s+[^\s]+\s+\([^)]+\)/g, "") // stack frames like `at fn (file.ts:1:1)`
        .replace(/\s{2,}/g, " ")
        .trim();
      if (stripped === "") return fallback;
      return stripped.length > MAX_USER_MESSAGE_LEN
        ? stripped.slice(0, MAX_USER_MESSAGE_LEN - 1) + "…"
        : stripped;
    }

    function extractBackendMessage(body: unknown): string | undefined {
      if (body && typeof body === "object" && "message" in body && typeof (body as { message: unknown }).message === "string") {
        return (body as { message: string }).message;
      }
      return undefined;
    }

    function extractValidationFields(body: unknown): Record<string, string> | undefined {
      if (body && typeof body === "object" && "fields" in body) {
        const fields = (body as { fields: unknown }).fields;
        if (fields && typeof fields === "object" && !Array.isArray(fields)) {
          const out: Record<string, string> = {};
          for (const [k, v] of Object.entries(fields as Record<string, unknown>)) {
            if (typeof v === "string") out[k] = v;
          }
          if (Object.keys(out).length > 0) return out;
        }
      }
      return undefined;
    }

    function defaultMessageForStatus(status: number): string {
      if (status === 401) return "Your session expired. Please sign in again.";
      if (status === 403) return "You don't have permission to do that.";
      if (status === 404) return "We couldn't find what you were looking for.";
      if (status >= 400 && status < 500) return "Your request couldn't be processed.";
      return "Something went wrong on our end. Please try again.";
    }
    ```

    **8) `classifyError(response)` — translates a non-2xx Response into an ApiError.** Defensive `try { body = await response.json(); } catch {}` because per CONTEXT assumption 2 some 5xx responses may be non-JSON (API Gateway 502 HTML).

    ```ts
    async function classifyError(response: Response, cause?: Error): Promise<ApiError> {
      const status = response.status;
      let body: unknown = null;
      try {
        body = await response.json();
      } catch {
        // Non-JSON response (e.g. API Gateway 502 HTML). body stays null; we'll use the default message.
      }
      const fallback = defaultMessageForStatus(status);
      const message = sanitizeMessage(extractBackendMessage(body), fallback);

      if (status === 401) return { kind: "unauthorized", message };
      if (status === 403) return { kind: "forbidden", status: 403, message };
      if (status >= 400 && status < 500) {
        const fields = extractValidationFields(body);
        const validation: { kind: "validation"; status: number; message: string; fields?: Record<string, string> } =
          fields ? { kind: "validation", status, message, fields } : { kind: "validation", status, message };
        return validation;
      }
      const server: { kind: "server"; status: number; message: string; cause?: Error } =
        cause ? { kind: "server", status, message, cause } : { kind: "server", status, message };
      return server;
    }
    ```

    **9) Signal composition helper.** Wrapper creates its own AbortController for the timeout; if caller passes `opts.signal`, abort the wrapper's controller when the caller's signal aborts.

    ```ts
    function composeSignals(timeoutMs: number, callerSignal?: AbortSignal): { signal: AbortSignal; cancelTimeout: () => void } {
      const controller = new AbortController();
      const timer = setTimeout(() => controller.abort(new Error("Request timed out")), timeoutMs);
      if (callerSignal) {
        if (callerSignal.aborted) {
          controller.abort(callerSignal.reason);
        } else {
          callerSignal.addEventListener("abort", () => controller.abort(callerSignal.reason), { once: true });
        }
      }
      const cancelTimeout = () => clearTimeout(timer);
      return { signal: controller.signal, cancelTimeout };
    }
    ```

    **10) Core `request<T>` private function** — used by `apiGet` and `apiPost`. Flow: get headers → if null, fire onUnauthorized + return unauthorized. Otherwise compose signal, fetch, classify on non-2xx (fire onUnauthorized on 401), parse JSON on 2xx, return `{ ok: true, data }`. All network/parse errors become `kind: "network"`. The caller's headers are NEVER merged — only the wrapper's Authorization header is sent (plus Content-Type for POST/PUT).

    ```ts
    async function request<T>(
      method: "GET" | "POST",
      path: string,
      body: unknown,
      opts: RequestOptions = {},
    ): Promise<Result<T, ApiError>> {
      const headers = await authorizedHeaders();
      if (!headers) {
        fireUnauthorized();
        return { ok: false, error: { kind: "unauthorized", message: defaultMessageForStatus(401) } };
      }

      const finalHeaders: Record<string, string> = { ...headers };
      if (method === "POST") finalHeaders["Content-Type"] = "application/json";

      const timeoutMs = opts.timeoutMs ?? DEFAULT_TIMEOUT_MS;
      const { signal, cancelTimeout } = composeSignals(timeoutMs, opts.signal);

      const url = `${getBaseUrl()}${path.startsWith("/") ? path : `/${path}`}`;

      let response: Response;
      try {
        response = await fetch(url, {
          method,
          headers: finalHeaders,
          body: method === "POST" ? JSON.stringify(body) : undefined,
          signal,
        });
      } catch (err) {
        cancelTimeout();
        const cause = err instanceof Error ? err : new Error(String(err));
        const isAbort = cause.name === "AbortError" || cause.message === "Request timed out";
        const baseError = {
          kind: "network" as const,
          message: isAbort ? "Request timed out. Please try again." : "Network error. Please check your connection.",
        };
        const error: ApiError = { ...baseError, cause };
        return { ok: false, error };
      }
      cancelTimeout();

      if (!response.ok) {
        const error = await classifyError(response);
        if (error.kind === "unauthorized") fireUnauthorized();
        return { ok: false, error };
      }

      // Parse 2xx body. Empty body (204) returns `null as T` — Phase 13+ tightens this per endpoint.
      let data: T;
      try {
        const text = await response.text();
        data = (text === "" ? (null as unknown) : JSON.parse(text)) as T;
      } catch (err) {
        const cause = err instanceof Error ? err : new Error(String(err));
        const error: ApiError = {
          kind: "server",
          status: response.status,
          message: "We received an invalid response from the server.",
          cause,
        };
        return { ok: false, error };
      }
      return { ok: true, data };
    }
    ```

## Trust Boundaries (Plan 12-02 scope)

| Boundary | Description |
|----------|-------------|
| client → API Gateway v2 | All authenticated frontend → backend traffic crosses here. Plan 12-02 is the chokepoint. |
| Cognito SDK ↔ client.ts | `getSession()` returns the IdToken. SDK is trusted; wrapper trusts its `null` signal as "must re-authenticate." |
| caller (`init` opts) → wrapper | A caller can pass `signal` + `timeoutMs`. They CANNOT pass headers — `RequestOptions` does not include a `headers` field. |
| backend response body → ApiError.message | Untrusted input; sanitized before being surfaced to the user. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-01 | Information disclosure | `ApiError.message` & `ApiError.cause` | mitigate | `sanitizeMessage()` strips Authorization headers, stack-frame patterns, collapses whitespace, and caps length at 240 chars. Falls back to a hardcoded copy keyed by HTTP status when the backend message is missing or fails sanitization. `cause` is never read by `useApiErrorUx` (verified in Plan 12-03 / 12-04) — it's attached for upstream logging only. |
| T-12-02 | Tampering | Authorization header injection by caller | mitigate | `RequestOptions` type omits a `headers` field — there is no way for a caller to supply headers. The wrapper builds the headers object from scratch on every request: `{ Authorization: 'Bearer <IdToken>' }` (+ `Content-Type` for POST). Single source of truth. |
| T-12-03 | Authentication bypass | IdToken vs AccessToken confusion | mitigate | `authorizedHeaders()` reads only `session.IdToken`. Acceptance criteria includes `grep` assertion that `session.AccessToken` is NOT referenced anywhere in `client.ts`. |
| T-12-04 | Denial of service | 401 retry storm / concurrent getSession refresh | accept | Retry budget = 0 (CONTEXT D-01). Concurrent requests each await their own `getSession()`; the Cognito SDK caches the in-memory session and serializes refresh internally. If profiling reveals a refresh storm, the dedup-via-shared-promise mitigation is deferred to v2.1 per CONTEXT. |
| T-12-05 | Information disclosure / DoS | SSR-time wrapper call short-circuiting all requests | mitigate | `getSession()` has the SSR guard (`typeof window === 'undefined'` → resolves null). | T-12-06 | Information disclosure | Backend leaking stack traces / DB errors into `ApiError.message` | mitigate | `extractBackendMessage` reads ONLY the `message` field of a JSON body (not stringified body, not error objects). `sanitizeMessage` then strips stack frames + auth headers + long strings.
